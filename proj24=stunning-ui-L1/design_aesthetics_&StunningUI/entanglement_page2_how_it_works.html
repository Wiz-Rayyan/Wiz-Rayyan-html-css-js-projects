Quantum entanglement — page 2: how it works technically

Three binding mechanisms
Event
listener
A click or input listener on element A calls a handler that immediately sets the same property on element B. Zero setTimeout, zero debounce — synchronous propagation is what creates the "instant" feel.
Mutation
observer
MutationObserver watches a DOM node's attributes, childList, or characterData. When A changes (even from a third-party script), the callback fires and mirrors the delta to B — without A knowing B exists.
Scroll
sync
A scroll listener on element A reads scrollTop / scrollHeight as a 0–1 ratio, then sets scrollTop = ratio × B.scrollHeight. This handles differing content heights gracefully. A guard flag prevents the echo-loop.

// Bind a pair of elements — click A animates B instantly function entangle(a, b) { const sync = (source, target) => { source.addEventListener('click', () => { target.classList.toggle('active'); target.dispatchEvent(new Event('entangled')); }); }; sync(a, b); sync(b, a); // bidirectional }
// Watch A for attribute changes, mirror to B function entangleAttr(a, b, attr) { const obs = new MutationObserver(mutations => { mutations.forEach(m => { if (m.attributeName === attr) { const val = a.getAttribute(attr); b.setAttribute(attr, val); // instant mirror } }); }); obs.observe(a, { attributes: true }); }
// Sync scroll position as a ratio (handles height diff) let syncing = false; function syncScroll(a, b) { a.addEventListener('scroll', () => { if (syncing) return; syncing = true; const ratio = a.scrollTop / (a.scrollHeight - a.clientHeight); b.scrollTop = ratio * (b.scrollHeight - b.clientHeight); syncing = false; }); }
// Mirror a CSS transform — rotation, scale, translate function mirrorTransform(a, b) { const obs = new MutationObserver(() => { const t = getComputedStyle(a).transform; const [,a,b2,c,d] = t.match(/matrix\((.+)\)/) ?.map(Number) ?? []; // Negate rotation to create mirror effect b.style.transform = `matrix(${a},${-b2},${-c},${d},0,0)`; }); obs.observe(a, { attributeFilter: ['style'] }); }
