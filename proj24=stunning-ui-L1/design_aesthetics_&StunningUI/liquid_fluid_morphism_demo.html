/* Irregular border-radius — 8-value shorthand */
border-radius: 60% 40% 50% 50% / 40% 50% 60% 50%;

/* Animate between shapes */
@keyframes morph {
  0%   { border-radius: 60% 40% 50% 50% / 40% 50% 60% 50%; }
  50%  { border-radius: 40% 60% 60% 40% / 60% 40% 50% 50%; }
  100% { border-radius: 60% 40% 50% 50% / 40% 50% 60% 50%; }
}

/* SVG goo filter for merged blobs */
feGaussianBlur stdDeviation="12"
